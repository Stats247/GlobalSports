Google Sheets setup (Apps Script)
  1. Create a Google Sheet and add a worksheet named blacklist.
  2. Open Extensions → Apps Script, paste the sample code below, then deploy as a Web app. Set access to the audience allowed by your use case and copy the deployed URL ending in /exec.
  3. Paste the URL into the Apps Script URL field under this panel and select Save URL. Test with Sync from Google Sheets.

Use an appropriately permissioned deployment. Anyone with the URL may be able to read or alter data if you deploy it publicly, so do not store sensitive information in this sheet.

const SHEET_NAME = 'blacklist';
function doGet(e) {
  const action = (e.parameter.action || 'read').toLowerCase();
  const sheet = getSheet_();
  if (action === 'read') {
    const values = sheet.getDataRange().getValues();
    const rows = values.slice(1).filter(r => r[0]).map(r => ({
      team: String(r[0]), reason: String(r[1] || ''),
      addedAt: String(r[2] || ''), active: String(r[3]).toLowerCase() !== 'false'
    }));
    return json_({ ok: true, teams: rows });
  }
  return json_({ ok: false, error: 'Unknown action' });
}
function doPost(e) {
  const body = JSON.parse((e.postData && e.postData.contents) || '{}');
  const sheet = getSheet_();
  const team = String(body.team || '').trim();
  if (body.action === 'write' && team) {
    const rows = sheet.getDataRange().getValues();
    const match = rows.findIndex((r, i) => i > 0 && String(r[0]).toLowerCase() === team.toLowerCase());
    const record = [team, String(body.reason || ''), new Date().toISOString(), body.active !== false];
    if (match > 0) sheet.getRange(match + 1, 1, 1, 4).setValues([record]);
    else sheet.appendRow(record);
    return json_({ ok: true });
  }
  if (body.action === 'delete' && team) {
    const rows = sheet.getDataRange().getValues();
    const match = rows.findIndex((r, i) => i > 0 && String(r[0]).toLowerCase() === team.toLowerCase());
    if (match > 0) sheet.deleteRow(match + 1);
    return json_({ ok: true });
  }
  return json_({ ok: false, error: 'Invalid action or missing team' });
}
function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);
  if (sh.getLastRow() === 0) sh.appendRow(['team', 'reason', 'addedAt', 'active']);
  return sh;
}
function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
